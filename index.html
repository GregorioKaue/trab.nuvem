// Pesquisa ferramentas pelo nome e pela descrição.
const pesquisa = document.getElementById("pesquisa"); // Obtém o campo de pesquisa da página.
const recursos = [ // Cria uma lista com os elementos que representam cada card.
...document.querySelectorAll("#listaRecursos > div") // Seleciona os cards dentro da lista de recursos.
];
pesquisa.addEventListener("input", () => { // Executa a pesquisa sempre que o texto muda.
const termo = pesquisa.value.toLocaleLowerCase("pt-BR"); // Lê o texto digitado e converte para minúsculas.
let encontrados = 0; // Inicializa a contagem de cards que correspondem à pesquisa.
recursos.forEach(item => { // Percorre todos os cards de ferramentas.
const corresponde = item.textContent // Lê o texto visível do card.
.toLocaleLowerCase("pt-BR").includes(termo); // Verifica se o card contém o termo pesquisado.
item.hidden = !corresponde; // Oculta cards que não correspondem ao termo.
if (corresponde) encontrados++; // Soma um resultado quando há correspondência.
});
document.getElementById("semResultados").hidden = // Obtém a mensagem de ausência de resultados.
encontrados !== 0; // Exibe a mensagem somente quando nenhum card corresponde.
});
// Exibe um aviso para os recursos que ainda não têm endereço configurado.
// O card Videoconferências não usa esta função: seu link abre o Google Meet diretamente.
function abrirRecurso(evento, nome) {
evento.preventDefault(); // Impede o comportamento padrão do link com href="#" para os recursos não configurados.
alert("Configure o endereço oficial de: " + nome); // Informa qual recurso precisa de um endereço oficial.
}
// Demonstração de seleção de documentos locais.
document.getElementById("arquivos") // Obtém o seletor de arquivos da página.
.addEventListener("change", evento => { // Executa quando o usuário seleciona ou altera arquivos.
const lista = document.getElementById("listaArquivos"); // Obtém a lista onde os arquivos serão mostrados.
lista.replaceChildren(); // Limpa os itens exibidos anteriormente.
[...evento.target.files].forEach(arquivo => { // Percorre cada arquivo selecionado.
const item = document.createElement("li"); // Cria um item de lista para o arquivo.
item.className = // Define classes do Bootstrap para organizar o item.
"list-group-item d-flex justify-content-between";
const nome = document.createElement("span"); // Cria um elemento para o nome do arquivo.
nome.textContent = arquivo.name; // Insere o nome do arquivo como texto.
const tamanho = document.createElement("small"); // Cria um elemento para exibir o tamanho.
tamanho.className = "text-muted"; // Aplica uma cor discreta ao tamanho.
tamanho.textContent = // Calcula e formata o tamanho do arquivo em kilobytes.
(arquivo.size / 1024).toFixed(1) + " KB";
item.append(nome, tamanho); // Coloca o nome e o tamanho dentro do item da lista.
lista.appendChild(item); // Adiciona o item à lista visível da página.
});
});
